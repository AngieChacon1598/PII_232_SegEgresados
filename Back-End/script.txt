Base de Datos para Productos 
------------------------------------------------ 

-- Crear la tabla 'product'
CREATE TABLE product (
    product_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY NOT NULL,                    
    name VARCHAR2(150) NOT NULL UNIQUE,                  -- Nombre del producto, único y no nulo
    description VARCHAR2(200),                           -- Descripción del producto
    price NUMBER(10, 2) NOT NULL CHECK (price > 0),      -- Precio, debe ser mayor que 0
    state CHAR(1) NOT NULL CHECK (state IN ('A', 'I'))    -- Estado ('A' = activo, 'I' = inactivo)
);

-- Insertar un nuevo producto (no es necesario especificar el product_id, se asignará automáticamente)
INSERT INTO product (name, description, price, state) 
VALUES ('Producto A', 'Descripción del producto A', 100.5, 'A');

-- Verificar que los datos se han insertado correctamente
SELECT * FROM product;

-- Crear tabla detalle_egresado con columna estado
CREATE TABLE detalle_egresado (
  ID_DETALLE           NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  CODIGO_EGRESADO      VARCHAR2(20) NOT NULL,
  FECHA_EGRESO         DATE,
  EMPRESA_ACTUAL       VARCHAR2(100),
  CARGO_ACTUAL         VARCHAR2(100),
  PAIS_RESIDENCIA      VARCHAR2(50),
  CIUDAD_RESIDENCIA    VARCHAR2(50),
  FECHA_INCORPORACION  DATE,
  AREA_TRABAJO         VARCHAR2(50),
  SUELDO_ACTUAL        NUMBER(10, 2),
  ESTADO               CHAR(1) NOT NULL CHECK (ESTADO IN ('A', 'I')) DEFAULT 'A',
  CONSTRAINT fk_detalle_egresado FOREIGN KEY (CODIGO_EGRESADO) REFERENCES EGRESADO(CODIGO)
);

CREATE TABLE certificacion (
    id_certificacion NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    codigo_egresado VARCHAR2(20) NOT NULL,
    nombre VARCHAR2(150) NOT NULL,
    institucion VARCHAR2(150) NOT NULL,
    fecha_obtencion DATE NOT NULL,
    archivo VARCHAR2(255),
    estado CHAR(1) DEFAULT 'A' CHECK (estado IN ('A', 'I')),
    CONSTRAINT fk_certificacion_egresado FOREIGN KEY (codigo_egresado) REFERENCES egresado(codigo)
);
